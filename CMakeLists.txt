cmake_minimum_required(VERSION 3.24)
project(JobDispatcherSP VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JobDispatcherSP requires Windows x64")
endif()
set(JD_SCS_SDK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/scs-sdk" CACHE PATH "SCS SDK 1.16")
add_library(JobDispatcherSP SHARED src/host.cpp src/dispatch.cpp src/engine_route.cpp src/labels.cpp src/exports.def)
set_target_properties(JobDispatcherSP PROPERTIES PREFIX "" OUTPUT_NAME "JobDispatcherSP")
target_include_directories(JobDispatcherSP PRIVATE src "${JD_SCS_SDK}/include")
target_compile_definitions(JobDispatcherSP PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(JobDispatcherSP PRIVATE version bcrypt psapi)
if(MINGW)
    target_compile_options(JobDispatcherSP PRIVATE -Wall -Wextra -Wno-macro-redefined)
    target_link_options(JobDispatcherSP PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(cmake/Overlay.cmake)
jd_add_overlay(JobDispatcherSP)
include(CTest)
if(BUILD_TESTING)

    if(MINGW)

    endif()
    add_test(NAME display-units-and-messages COMMAND jd-display-units-test)
