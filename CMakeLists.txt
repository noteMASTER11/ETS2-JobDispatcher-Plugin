cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" JD_RELEASE_VERSION LIMIT_COUNT 1)
project(JobDispatcherSP VERSION "${JD_RELEASE_VERSION}.0" LANGUAGES C CXX)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "JobDispatcherSP requires Windows x64")
endif()
set(JD_SCS_SDK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/scs-sdk" CACHE PATH "SCS SDK 1.16")
add_library(JobDispatcherSP SHARED src/host.cpp src/dispatch.cpp src/engine_route.cpp src/labels.cpp src/offline_assets.cpp src/exports.def)
set_target_properties(JobDispatcherSP PROPERTIES PREFIX "" OUTPUT_NAME "JobDispatcherSP")
enable_language(RC)
configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
target_sources(JobDispatcherSP PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_include_directories(JobDispatcherSP PRIVATE src "${JD_SCS_SDK}/include")
target_compile_definitions(JobDispatcherSP PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(JobDispatcherSP PRIVATE version bcrypt psapi shell32 ole32)
if(MINGW)
    target_compile_options(JobDispatcherSP PRIVATE -Wall -Wextra -Wno-macro-redefined)
    target_link_options(JobDispatcherSP PRIVATE -static -static-libgcc -static-libstdc++)
endif()
